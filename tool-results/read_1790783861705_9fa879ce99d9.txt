    67→    67→/** meters → feet/inches string, e.g. 2.4 m → 7′10″ */
    68→    68→function feetInches(m: number): string {
    69→    69→  const totalIn = m * 39.3701
    70→    70→  const ft = Math.floor(totalIn / 12)
    71→    71→  let inch = Math.round(totalIn - ft * 12)
    72→    72→  let ftOut = ft
    73→    73→  if (inch === 12) {
    74→    74→    ftOut += 1
    75→    75→    inch = 0
    76→    76→  }
    77→    77→  return `${ftOut}′${inch}″`
    78→    78→}

   183→   183→function CameraRig({
   184→   184→  roomW, roomD, viewMode, controlsRef, animatingRef,
   185→   185→}: {
   186→   186→  roomW: number
   187→   187→  roomD: number
   188→   188→  viewMode: 'plan' | '3d'
   189→   189→  controlsRef: React.MutableRefObject<any>
   190→   190→  animatingRef: React.MutableRefObject<boolean>
   191→   191→}) {
   192→   192→  const zoomReq = usePlanner((s) => s.zoomReq)
   193→   193→  const lastReq = useRef(0)
   194→   194→  /** active dolly animation: camera position we are easing toward, or null */
   195→   195→  const dollyTarget = useRef<THREE.Vector3 | null>(null)
   196→   196→
   197→   197→  const target = useMemo(() => {
   198→   198→    return viewMode === 'plan'
   199→   199→      ? { pos: new THREE.Vector3(0.02, Math.max(roomW, roomD) * 1.3 + 3, 0.02), look: new THREE.Vector3(0, 0, 0) }
   200→   200→      : { pos: new THREE.Vector3(roomW * 0.52, Math.max(6, Math.max(roomW, roomD) * 0.52), roomD * 0.82), look: new THREE.Vector3(0, 0.4, 0) }
   201→   201→  }, [viewMode, roomW, roomD])
   202→   202→
   203→   203→  useEffect(() => {
   204→   204→    animatingRef.current = true
   205→   205→  }, [target, animatingRef])
   206→   206→
   207→   207→  // react to zoom requests coming from the HUD buttons / keyboard
   208→   208→  useEffect(() => {
   209→   209→    if (zoomReq.n === lastReq.current) return
   210→   210→    lastReq.current = zoomReq.n
   211→   211→    const c = controlsRef.current
   212→   212→    if (!c) return
   213→   213→    if (zoomReq.kind === 'fit') {
   214→   214→      // re-frame the room: replay the swing to the current view's home position
   215→   215→      animatingRef.current = true
   216→   216→      return
   217→   217→    }
   218→   218→    const cam = c.object as THREE.Camera
   219→   219→    const dir = cam.position.clone().sub(c.target)
   220→   220→    const len = dir.length()
   221→   221→    const next = THREE.MathUtils.clamp(
   222→   222→      zoomReq.kind === 'in' ? len * 0.72 : len * 1.42,
   223→   223→      c.minDistance ?? 3.5,
   224→   224→      c.maxDistance ?? 48,
   225→   225→    )
   226→   226→    if (Math.abs(next - len) < 0.05) return
   227→   227→    dollyTarget.current = c.target.clone().add(dir.setLength(next))
   228→   228→  }, [zoomReq, controlsRef])
   229→   229→
   230→   230→  useFrame((state, dt) => {
   231→   231→    const c = controlsRef.current
   232→   232→    if (!c) return
   233→   233→
   234→   234→    // eased zoom-in/out along the view vector
   235→   235→    if (dollyTarget.current) {
   236→   236→      damp3(state.camera.position, dollyTarget.current, 0.16, dt)
   237→   237→      if (state.camera.position.distanceTo(dollyTarget.current) < 0.04) {
   238→   238→        state.camera.position.copy(dollyTarget.current)
   239→   239→        dollyTarget.current = null
   240→   240→      }
   241→   241→      return
   242→   242→    }
   243→   243→
   244→   244→    if (animatingRef.current) {
   245→   245→      c.enabled = false
   246→   246→      // top-down needs a stable roll: screen-up = north (-z); 3D restores standard up
   247→   247→      if (viewMode === 'plan') state.camera.up.set(0, 0, -1)
   248→   248→      else state.camera.up.set(0, 1, 0)
   249→   249→      damp3(state.camera.position, target.pos, 0.42, dt)
   250→   250→      damp3(c.target, target.look, 0.42, dt)
   251→   251→      state.camera.lookAt(target.look)
   252→   252→      if (state.camera.position.distanceTo(target.pos) < 0.06) {
   253→   253→        animatingRef.current = false
   254→   254→        c.enabled = true
   255→   255→      }
   256→   256→    }
   257→   257→  })
   258→   258→  return null
   259→   259→}

   398→   398→function EgressPaths({ w, d }: { w: number; d: number }) {
   399→   399→  const { curves, counts, total } = useMemo(() => {
   400→   400→    const v = (x: number, z: number) => new THREE.Vector3(x, 0.035, z)
   401→   401→    const mk = (pts: THREE.Vector3[]) => new THREE.CatmullRomCurve3(pts, false, 'catmullrom', 0.4)
   402→   402→    const cs = egressRoutes({ w, d }).map((r) => mk(r.map((p) => v(p.x, p.z))))
   403→   403→    const countsArr = cs.map((c) => Math.max(7, Math.floor(c.getLength() / 0.62)))
   404→   404→    return { curves: cs, counts: countsArr, total: countsArr.reduce((a, b) => a + b, 0) }
   405→   405→  }, [w, d])
   406→   406→
   407→   407→  const groupRef = useRef<THREE.Group>(null)
   408→   408→
   409→   409→  useFrame(({ clock }) => {
   410→   410→    const t0 = clock.elapsedTime * 0.16
   411→   411→    let idx = 0
   412→   412→    for (let ci = 0; ci < curves.length; ci++) {
   413→   413→      const c = curves[ci]
   414→   414→      const n = counts[ci]
   415→   415→      for (let i = 0; i < n; i++, idx++) {
   416→   416→        const obj = groupRef.current?.children[idx]
   417→   417→        if (!obj) continue
   418→   418→        const t = (t0 + i / n) % 1
   419→   419→        const p = c.getPointAt(t)
   420→   420→        const tan = c.getTangentAt(t)
   421→   421→        obj.position.set(p.x, 0.035, p.z)
   422→   422→        obj.lookAt(p.x + tan.x, 0.035, p.z + tan.z)
   423→   423→        const s = 0.55 + 0.55 * Math.sin(t * Math.PI)
   424→   424→        obj.scale.setScalar(s)
   425→   425→      }
   426→   426→    }
   427→   427→  })
   428→   428→
   429→   429→  return (
   430→   430→    <group ref={groupRef}>
   431→   431→      {Array.from({ length: total }, (_, i) => (
   432→   432→        <Chevron key={i} />
   433→   433→      ))}
   434→   434→    </group>
   435→   435→  )
   436→   436→}

   440→   440→function PlacedItemView({
   441→   441→  uid, id, x, z, rot, brand, selected, dragging, interactive,
   442→   442→  onPointerDown, onOpenContext,
   443→   443→}: {
   444→   444→  uid: string
   445→   445→  id: string
   446→   446→  x: number
   447→   447→  z: number
   448→   448→  rot: number
   449→   449→  brand: string
   450→   450→  selected: boolean
   451→   451→  dragging: boolean
   452→   452→  /** false while the measure tool is armed — no grab cursor / selection styling */
   453→   453→  interactive: boolean
   454→   454→  onPointerDown: (e: ThreeEvent<PointerEvent>, uid: string) => void
   455→   455→  onOpenContext: (e: ThreeEvent<MouseEvent>, uid: string) => void
   456→   456→}) {
   457→   457→  const groupRef = useRef<THREE.Group>(null)
   458→   458→  const ringRef = useRef<THREE.Mesh>(null)
   459→   459→  useFrame((_, dt) => {
   460→   460→    if (groupRef.current) damp(groupRef.current.rotation, 'y', rot, 0.16, dt)
   461→   461→    // selected ring gently breathes so the selection feels alive
   462→   462→    if (ringRef.current && selected) {
   463→   463→      const s = 1 + Math.sin(performance.now() * 0.004) * 0.025
   464→   464→      ringRef.current.scale.setScalar(s)
   465→   465→    }
   466→   466→  })
   467→   467→  const cat = getItem(id)
   468→   468→  const { hx, hz } = halfExtents(id, rot)
   469→   469→  const ringR = Math.max(hx, hz) + 0.14
   470→   470→
   471→   471→  return (
   472→   472→    <>
   473→   473→      {selected && (
   474→   474→        <mesh ref={ringRef} rotation={[-Math.PI / 2, 0, 0]} position={[x, 0.014, z]}>
   475→   475→          <ringGeometry args={[ringR, ringR + 0.06, 44]} />
   476→   476→          <meshBasicMaterial color={brand} transparent opacity={0.85} />
   477→   477→        </mesh>
   478→   478→      )}
   479→   479→      {(dragging || selected) && cat && (
   480→   480→        <mesh rotation={[-Math.PI / 2, 0, 0]} position={[x, 0.012, z]}>
   481→   481→          <planeGeometry args={[hx * 2, hz * 2]} />
   482→   482→          <meshBasicMaterial color={brand} transparent opacity={dragging ? 0.22 : 0.1} depthWrite={false} />
   483→   483→        </mesh>
   484→   484→      )}
   485→   485→      <group
   486→   486→        ref={groupRef}
   487→   487→        position={[x, 0, z]}
   488→   488→        onPointerDown={(e) => onPointerDown(e, uid)}
   489→   489→        onContextMenu={(e) => onOpenContext(e, uid)}
   490→   490→        onPointerOver={() => {
   491→   491→          if (interactive) document.body.style.cursor = 'grab'
   492→   492→        }}
   493→   493→        onPointerOut={() => {
   494→   494→          document.body.style.cursor = ''
   495→   495→        }}
   496→   496→      >
   497→   497→        <ItemModel id={id} brand={brand} />
   498→   498→      </group>
   499→   499→    </>
   500→   500→  )
   501→   501→}

   516→   516→  )

   521→   521→export default function Scene() {
   522→   522→  const room = usePlanner((s) => s.room)
   523→   523→  const items = usePlanner((s) => s.items)
   524→   524→  const brand = usePlanner((s) => s.brandColor)
   525→   525→  const viewMode = usePlanner((s) => s.viewMode)
   526→   526→  const showFire = usePlanner((s) => s.showFire)
   527→   527→  const zones = usePlanner((s) => s.zones)
   528→   528→  const showZones = usePlanner((s) => s.showZones)
   529→   529→  const armedId = usePlanner((s) => s.armedId)
   530→   530→  const selectedUid = usePlanner((s) => s.selectedUid)
   531→   531→  const addItem = usePlanner((s) => s.addItem)
   532→   532→  const moveItem = usePlanner((s) => s.moveItem)
   533→   533→  const select = usePlanner((s) => s.select)
   534→   534→  const arm = usePlanner((s) => s.arm)
   535→   535→  const openContextMenu = usePlanner((s) => s.openContextMenu)
   536→   536→  const measureMode = usePlanner((s) => s.measureMode)
   537→   537→  const measureA = usePlanner((s) => s.measureA)
   538→   538→  const measureB = usePlanner((s) => s.measureB)
   539→   539→  const setMeasurePoint = usePlanner((s) => s.setMeasurePoint)
   540→   540→  const { resolvedTheme } = useTheme()
   541→   541→  const dark = resolvedTheme === 'dark'
   542→   542→  const pal = dark ? NIGHT : DAY
   543→   543→
   544→   544→  const controlsRef = useRef<any>(null)
   545→   545→  const animatingRef = useRef(true)
   546→   546→  const dragUid = useRef<string | null>(null)
   547→   547→  const downScreen = useRef<{ x: number; y: number } | null>(null)
   548→   548→  /** where the right-button went down, so a right-drag pan doesn't open the menu */
   549→   549→  const rightDown = useRef<{ x: number; y: number } | null>(null)
   550→   550→  const [ghost, setGhost] = useState<{ x: number; z: number } | null>(null)
   551→   551→  const [draggingUid, setDraggingUid] = useState<string | null>(null)
   552→   552→  /** live cursor point while the measure tool is armed (preview line) */
   553→   553→  const [measureHover, setMeasureHover] = useState<{ x: number; z: number } | null>(null)
   554→   554→
   555→   555→  useEffect(() => {
   556→   556→    const up = () => {
   557→   557→      dragUid.current = null
   558→   558→      setDraggingUid(null)
   559→   559→      downScreen.current = null
   560→   560→      if (controlsRef.current) controlsRef.current.enabled = true
   561→   561→    }
   562→   562→    window.addEventListener('pointerup', up)
   563→   563→    return () => window.removeEventListener('pointerup', up)
   564→   564→  }, [])
   565→   565→
   566→   566→  const onGroundDown = (e: ThreeEvent<PointerEvent>) => {
   567→   567→    if (e.nativeEvent.button === 2) {
   568→   568→      rightDown.current = { x: e.nativeEvent.clientX, y: e.nativeEvent.clientY }
   569→   569→      return
   570→   570→    }
   571→   571→    downScreen.current = { x: e.nativeEvent.clientX, y: e.nativeEvent.clientY }
   572→   572→  }
   573→   573→
   574→   574→  const onGroundMove = (e: ThreeEvent<PointerEvent>) => {
   575→   575→    const pt = floorPoint(e)
   576→   576→    if (dragUid.current) {
   577→   577→      moveItem(dragUid.current, pt.x, pt.z)
   578→   578→      return
   579→   579→    }
   580→   580→    if (measureMode) {
   581→   581→      // only re-render when the cursor moved meaningfully (2 cm)
   582→   582→      setMeasureHover((prev) =>
   583→   583→        !prev || Math.abs(prev.x - pt.x) > 0.02 || Math.abs(prev.z - pt.z) > 0.02 ? { x: pt.x, z: pt.z } : prev,
   584→   584→      )
   585→   585→      return
   586→   586→    }
   587→   587→    if (armedId) {
   588→   588→      const p = clampToRoom(pt.x, pt.z, armedId, 0, room)
   589→   589→      setGhost(p)
   590→   590→    }
   591→   591→  }
   592→   592→
   593→   593→  const onGroundUp = (e: ThreeEvent<PointerEvent>) => {
   594→   594→    const ds = downScreen.current
   595→   595→    downScreen.current = null
   596→   596→    if (dragUid.current) {
   597→   597→      dragUid.current = null
   598→   598→      setDraggingUid(null)
   599→   599→      if (controlsRef.current) controlsRef.current.enabled = true
   600→   600→      return
   601→   601→    }
   602→   602→    if (!ds) return
   603→   603→    const moved = Math.hypot(e.nativeEvent.clientX - ds.x, e.nativeEvent.clientY - ds.y)
   604→   604→    if (moved > 7) return
   605→   605→    if (measureMode) {
   606→   606→      const pt = floorPoint(e)
   607→   607→      setMeasurePoint(pt)
   608→   608→      setMeasureHover({ x: pt.x, z: pt.z })
   609→   609→      return
   610→   610→    }
   611→   611→    if (armedId) {
   612→   612→      const pt = floorPoint(e)
   613→   613→      addItem(armedId, pt.x, pt.z)
   614→   614→      arm(null)
   615→   615→      setGhost(null)
   616→   616→    } else {
   617→   617→      select(null)
   618→   618→    }
   619→   619→  }
   620→   620→
   621→   621→  const onItemDown = (e: ThreeEvent<PointerEvent>, uid: string) => {
   622→   622→    if (measureMode) return // measure clicks pass through pieces to the floor
   623→   623→    if (armedId) return // let placement pass through
   624→   624→    if (e.nativeEvent.button !== 0) return // right/middle click → context menu, not drag
   625→   625→    e.stopPropagation()
   626→   626→    select(uid)
   627→   627→    dragUid.current = uid
   628→   628→    setDraggingUid(uid)
   629→   629→    downScreen.current = null
   630→   630→    if (controlsRef.current) controlsRef.current.enabled = false
   631→   631→  }
   632→   632→
   633→   633→  const onItemContext = (e: ThreeEvent<MouseEvent>, uid: string) => {
   634→   634→    if (measureMode) return // no context menu while measuring
   635→   635→    e.stopPropagation()
   636→   636→    e.nativeEvent.preventDefault()
   637→   637→    // ignore if this was the end of a right-drag pan
   638→   638→    const rd = rightDown.current
   639→   639→    rightDown.current = null
   640→   640→    if (rd && Math.hypot(e.nativeEvent.clientX - rd.x, e.nativeEvent.clientY - rd.y) > 6) return
   641→   641→    openContextMenu(uid, e.nativeEvent.clientX, e.nativeEvent.clientY)
   642→   642→  }
   643→   643→
   644→   644→  return (
   645→   645→    <Canvas
   646→   646→      shadows
   647→   647→      dpr={[1, 2]}
   648→   648→      camera={{ position: [11, 8.5, 11], fov: 40, near: 0.1, far: 150 }}
   649→   649→      gl={{ antialias: true }}
   650→   650→      className={`!absolute inset-0${measureMode ? ' cursor-crosshair' : ''}`}
   651→   651→    >
   652→   652→      <color attach="background" args={[pal.bg]} />
   653→   653→      <fog attach="fog" args={[pal.fog, 34, 85]} />
   654→   654→
   655→   655→      <ambientLight intensity={pal.ambient} />
   656→   656→      <hemisphereLight args={[pal.hemiSky, pal.hemiGround, pal.hemi]} />
   657→   657→      <directionalLight
   658→   658→        position={[9, 13, 7]}
   659→   659→        intensity={pal.sun}
   660→   660→        castShadow
   661→   661→        shadow-mapSize={[2048, 2048]}
   662→   662→        shadow-camera-left={-14}
   663→   663→        shadow-camera-right={14}
   664→   664→        shadow-camera-top={14}
   665→   665→        shadow-camera-bottom={-14}
   666→   666→        shadow-camera-near={1}
   667→   667→        shadow-camera-far={45}
   668→   668→        shadow-bias={-0.0004}
   669→   669→      />
   670→   670→
   671→   671→      <Suspense fallback={null}>
   672→   672→        <Room w={room.w} d={room.d} showFire={showFire} pal={pal} />
   673→   673→        {showZones && <ZoneWashes zones={drawableZones(zones, room)} dark={dark} />}
   674→   674→        {showFire && <EgressPaths w={room.w} d={room.d} />}
   675→   675→
   676→   676→        {/* interaction ground */}
   677→   677→        <mesh
   678→   678→          position={[0, -0.09, 0]}
   679→   679→          rotation={[-Math.PI / 2, 0, 0]}
   680→   680→          receiveShadow
   681→   681→          onPointerDown={onGroundDown}
   682→   682→          onPointerMove={onGroundMove}
   683→   683→          onPointerUp={onGroundUp}
   684→   684→        >
   685→   685→          <planeGeometry args={[90, 90]} />
   686→   686→          <meshStandardMaterial color={pal.apron} roughness={1} />
   687→   687→        </mesh>
   688→   688→
   689→   689→        {items.map((it) => (
   690→   690→          <PlacedItemView
   691→   691→            key={it.uid}
   692→   692→            {...it}
   693→   693→            brand={brand}
   694→   694→            selected={selectedUid === it.uid && !measureMode}
   695→   695→            dragging={draggingUid === it.uid}
   696→   696→            interactive={!measureMode}
   697→   697→            onPointerDown={onItemDown}
   698→   698→            onOpenContext={onItemContext}
   699→   699→          />
   700→   700→        ))}
   701→   701→
   702→   702→        {armedId && ghost && <Ghost id={armedId} x={ghost.x} z={ghost.z} brand={brand} />}
   703→   703→
   704→   704→        {measureMode && measureA && (
   705→   705→          <MeasureGizmo a={measureA} b={measureB} hover={measureHover} />
   706→   706→        )}
   707→   707→      </Suspense>
   708→   708→
   709→   709→      <CameraRig
   710→   710→        roomW={room.w}
   711→   711→        roomD={room.d}
   712→   712→        viewMode={viewMode}
   713→   713→        controlsRef={controlsRef}
   714→   714→        animatingRef={animatingRef}
   715→   715→      />
   716→   716→      <OrbitControls
   717→   717→        ref={controlsRef}
   718→   718→        makeDefault
   719→   719→        enableDamping
   720→   720→        dampingFactor={0.09}
   721→   721→        minDistance={3.5}
   722→   722→        maxDistance={48}
   723→   723→        maxPolarAngle={Math.PI / 2 - 0.05}
   724→   724→        enableRotate={viewMode === '3d'}
   725→   725→        onStart={() => {
   726→   726→          animatingRef.current = false
   727→   727→        }}
   728→   728→      />
   729→   729→    </Canvas>
   730→   730→  )
   731→   731→}
