    41→/** Build and download a bill-of-materials CSV for the current layout. */
    42→function downloadBomCsv(
    43→  projectName: string, templateId: string, areaM2: number, seats: number,
    44→  lines: Array<{ name: string; qty: number; unit: number; total: number; custom?: boolean }>,
    45→  subtotal: number, install: number, contingency: number, total: number, budget: number | null,
    46→  caps: Array<{ label: string; cap: number; spend: number; status: 'under' | 'close' | 'over' }>,
    47→) {
    48→  const esc = (v: string | number) => {
    49→    const s = String(v)
    50→    return /[",\n]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s
    51→  }
    52→  const rows: string[] = []
    53→  rows.push('FORMA — Bill of materials')
    54→  rows.push(`Project,${esc(projectName || 'Untitled layout')}`)
    55→  rows.push(`Template,${esc(templateId === 'custom' ? 'Custom layout' : TEMPLATES[templateId as 'cafe']?.name ?? templateId)}`)
    56→  rows.push(`Floor area,${areaM2.toFixed(1)} m2`)
    57→  rows.push(`Seats,${seats}`)
    58→  rows.push(`Exported,${new Date().toISOString().slice(0, 10)}`)
    59→  rows.push('')
    60→  rows.push('Item,Qty,Unit price (USD),Line total (USD),Pricing')
    61→  for (const l of lines) {
    62→    rows.push([esc(l.name), l.qty, l.unit, l.total, l.custom ? 'custom' : 'catalog'].join(','))
    63→  }
    64→  rows.push('')
    65→  rows.push(`Subtotal,,,${subtotal},`)
    66→  rows.push(`Delivery & install (12%),,,${install},`)
    67→  rows.push(`Contingency (8%),,,${contingency},`)
    68→  rows.push(`Estimated total,,,${total},`)
    69→  if (budget !== null) rows.push(`Budget,,,${budget},`)
    70→  if (caps.length > 0) {
    71→    rows.push('')
    72→    rows.push('Category caps')
    73→    rows.push('Category,Cap (USD),Spend (USD),Status')
    74→    for (const c of caps) {
    75→      rows.push([esc(c.label), c.cap, c.spend, c.status.toUpperCase()].join(','))
    76→    }
    77→  }
    78→  const blob = new Blob([rows.join('\n')], { type: 'text/csv;charset=utf-8' })
    79→  const url = URL.createObjectURL(blob)
    80→  const a = document.createElement('a')
    81→  const slug = (projectName || 'layout').toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/(^-|-$)/g, '') || 'layout'
    82→  a.href = url
    83→  a.download = `forma-bom-${slug}.csv`
    84→  document.body.appendChild(a)
    85→  a.click()
    86→  a.remove()
    87→  setTimeout(() => URL.revokeObjectURL(url), 4000)
    88→}
