    39→/** Build and download a bill-of-materials CSV for the current layout. */
    40→function downloadBomCsv(
    41→  projectName: string, templateId: string, areaM2: number, seats: number,
    42→  lines: Array<{ name: string; qty: number; unit: number; total: number; custom?: boolean }>,
    43→  subtotal: number, install: number, contingency: number, total: number, budget: number | null,
    44→  caps: Array<{ label: string; cap: number; spend: number; status: 'under' | 'close' | 'over' }>,
    45→) {
    46→  const esc = (v: string | number) => {
    47→    const s = String(v)
    48→    return /[",\n]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s
    49→  }
    50→  const rows: string[] = []
    51→  rows.push('FORMA — Bill of materials')
    52→  rows.push(`Project,${esc(projectName || 'Untitled layout')}`)
    53→  rows.push(`Template,${esc(templateId === 'custom' ? 'Custom layout' : TEMPLATES[templateId as 'cafe']?.name ?? templateId)}`)
    54→  rows.push(`Floor area,${areaM2.toFixed(1)} m2`)
    55→  rows.push(`Seats,${seats}`)
    56→  rows.push(`Exported,${new Date().toISOString().slice(0, 10)}`)
    57→  rows.push('')
    58→  rows.push('Item,Qty,Unit price (USD),Line total (USD),Pricing')
    59→  for (const l of lines) {
    60→    rows.push([esc(l.name), l.qty, l.unit, l.total, l.custom ? 'custom' : 'catalog'].join(','))
    61→  }
    62→  rows.push('')
    63→  rows.push(`Subtotal,,,${subtotal},`)
    64→  rows.push(`Delivery & install (12%),,,${install},`)
    65→  rows.push(`Contingency (8%),,,${contingency},`)
    66→  rows.push(`Estimated total,,,${total},`)
    67→  if (budget !== null) rows.push(`Budget,,,${budget},`)
    68→  if (caps.length > 0) {
    69→    rows.push('')
    70→    rows.push('Category caps')
    71→    rows.push('Category,Cap (USD),Spend (USD),Status')
    72→    for (const c of caps) {
    73→      rows.push([esc(c.label), c.cap, c.spend, c.status.toUpperCase()].join(','))
    74→    }
    75→  }
    76→  const blob = new Blob([rows.join('\n')], { type: 'text/csv;charset=utf-8' })
    77→  const url = URL.createObjectURL(blob)
    78→  const a = document.createElement('a')
    79→  const slug = (projectName || 'layout').toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/(^-|-$)/g, '') || 'layout'
    80→  a.href = url
    81→  a.download = `forma-bom-${slug}.csv`
    82→  document.body.appendChild(a)
    83→  a.click()
    84→  a.remove()
    85→  setTimeout(() => URL.revokeObjectURL(url), 4000)
    86→}
